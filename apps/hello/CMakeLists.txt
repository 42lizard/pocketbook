cmake_minimum_required(VERSION 3.21)
get_filename_component(APP_NAME "${CMAKE_CURRENT_SOURCE_DIR}" NAME)
project(${APP_NAME} LANGUAGES CXX)
option(POCKETBOOK_SIMULATOR "Build with the shared PocketBook simulator" OFF)
if(POCKETBOOK_SIMULATOR)
  include(../../mk/pocketbook-simulator.cmake)
  pocketbook_add_simulator_app(${APP_NAME} SOURCES src/main.cpp QRC qml/hello.qrc)
  enable_testing()
  add_test(NAME hello-simulator COMMAND ${APP_NAME} --simulator-smoke-test)
  set_tests_properties(hello-simulator PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
  return()
endif()
include(../../mk/pocketbook-qt6.cmake)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../build/${APP_NAME}")
pocketbook_add_app(${APP_NAME} SOURCES src/main.cpp QRC qml/hello.qrc)
