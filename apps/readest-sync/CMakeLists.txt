cmake_minimum_required(VERSION 3.21)
project(readest-sync LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(READEST_HOST_TESTS "Build native core tests without Qt or the PocketBook SDK" OFF)
option(READEST_DESKTOP "Build desktop tests with device stubs" OFF)
option(POCKETBOOK_SIMULATOR "Build with the shared PocketBook simulator" OFF)
set(READEST_SIMULATOR ${POCKETBOOK_SIMULATOR})
set(POCKETBOOK_SIM_CONTROLS "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/pocketbook-simulator/qt-controls")
if(POCKETBOOK_SIMULATOR)
  include(../../mk/pocketbook-simulator.cmake)
  set(READEST_SIM_DIR "${POCKETBOOK_SIMULATOR_DIR}/apps/readest-sync")
endif()
if(READEST_SIMULATOR)
  set(READEST_DESKTOP ON)
endif()
if(READEST_DESKTOP)
  find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick)
  set(CMAKE_AUTOMOC ON)
  set(CMAKE_AUTORCC ON)
  set(DEVICE_SOURCE tests/platform_stub.cpp)
  if(READEST_SIMULATOR)
    set(DEVICE_SOURCE "${READEST_SIM_DIR}/simulator.cpp" "${READEST_SIM_DIR}/simulator.h" "${READEST_SIM_DIR}/simulator.qrc")
  endif()
endif()
if(READEST_DESKTOP OR READEST_HOST_TESTS)
  if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Host tests require a native compiler; unset CMAKE_TOOLCHAIN_FILE and use a separate build directory")
  endif()
  find_package(LibXml2 REQUIRED)
  find_package(OpenSSL REQUIRED)
  find_package(SQLite3 REQUIRED)
  if(TARGET SQLite3::SQLite3)
    set(SQLITE_TARGET SQLite3::SQLite3)
  else()
    set(SQLITE_TARGET SQLite::SQLite3) # CMake before 4.3
  endif()
  find_package(CURL REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
  find_package(Threads REQUIRED)
  set(DEPS LibXml2::LibXml2 OpenSSL::Crypto ${SQLITE_TARGET} CURL::libcurl PkgConfig::JSONC Threads::Threads ${CMAKE_DL_LIBS})
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
  # Assertions are executable test contracts even in Release configurations.
  add_compile_options(-UNDEBUG)
  enable_testing()
else()
  include(../../mk/pocketbook-qt6.cmake)
  set(DEVICE_SOURCE src/platform.cpp)
  set(DEPS "${POCKETBOOK_QT_SDK}/libcurl.so.4" "${POCKETBOOK_TARGET_LIB_DIR}/libxml2.so"
           "${POCKETBOOK_TARGET_LIB_DIR}/libcrypto.so" "${POCKETBOOK_TARGET_LIB_DIR}/libsqlite3.a"
           "${POCKETBOOK_TARGET_LIB_DIR}/libjson-c.a" pthread dl m)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../build/readest-sync")
endif()
add_library(readest-core STATIC
  src/application.cpp src/upload.cpp src/cloud.cpp src/download.cpp src/http.cpp src/integrity.cpp src/library.cpp
  src/position.cpp src/probe.cpp src/progress.cpp src/state.cpp src/sync.cpp
  src/vendor/miniz/miniz.c)
target_compile_definitions(readest-core PUBLIC MINIZ_NO_DEFLATE_APIS MINIZ_NO_ZLIB_APIS)
target_include_directories(readest-core PUBLIC src)
if(NOT READEST_DESKTOP AND NOT READEST_HOST_TESTS)
  target_include_directories(readest-core SYSTEM PUBLIC "${POCKETBOOK_TARGET_INCLUDE_DIR}" /usr/include/libxml2)
endif()
target_link_libraries(readest-core PUBLIC ${DEPS})
if(READEST_DESKTOP OR READEST_HOST_TESTS)
  target_compile_options(readest-core PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Werror>")
  add_subdirectory(tests)
endif()
set(PRESENTATION_SOURCES src/controller.cpp src/controller.h src/library_model.cpp src/library_model.h
  src/operation_runner.cpp src/device_adapter.cpp src/cover_loader.cpp)
if(NOT READEST_HOST_TESTS OR READEST_DESKTOP)
  add_library(readest-presentation STATIC ${PRESENTATION_SOURCES})
  set_target_properties(readest-presentation PROPERTIES AUTOMOC ON)
  target_link_libraries(readest-presentation PUBLIC readest-core Qt6::Quick Qt6::Qml Qt6::Gui Qt6::Core)
endif()
set(APP_SOURCES src/main.cpp ${DEVICE_SOURCE})
if(READEST_DESKTOP)
  add_executable(readest-sync ${APP_SOURCES} qml/readest.qrc)
  target_compile_definitions(readest-sync PRIVATE READEST_DESKTOP READEST_TEST_CONTROLS="${POCKETBOOK_SIM_CONTROLS}")
  target_link_libraries(readest-sync PRIVATE readest-presentation)
  add_executable(qt-ui-test tests/qt_ui_test.cpp tests/platform_stub.cpp qml/readest.qrc)
  target_compile_definitions(qt-ui-test PRIVATE READEST_DESKTOP READEST_TEST_CONTROLS="${POCKETBOOK_SIM_CONTROLS}")
  target_link_libraries(qt-ui-test PRIVATE readest-presentation)
  if(READEST_SIMULATOR)
    include("${READEST_SIM_DIR}/integrate.cmake")
  endif()
  add_executable(cover-loader-test tests/cover_loader_test.cpp)
  target_link_libraries(cover-loader-test PRIVATE readest-presentation)
  target_compile_options(cover-loader-test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME cover-loader COMMAND cover-loader-test)
  add_test(NAME qt-ui COMMAND qt-ui-test)
  set_tests_properties(qt-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
elseif(NOT READEST_HOST_TESTS)
  pocketbook_add_app(readest-sync SOURCES ${APP_SOURCES} QRC qml/readest.qrc LIBRARIES readest-presentation)
  target_link_options(readest-sync PRIVATE "-Wl,--exclude-libs,libsqlite3.a:libjson-c.a")
endif()
