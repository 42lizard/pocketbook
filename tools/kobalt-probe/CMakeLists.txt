cmake_minimum_required(VERSION 3.21)
project(kobalt-probe LANGUAGES CXX)
option(KOBALT_PROBE_HOST "Run the compatibility probe on the build host" OFF)
set(KOBALT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../external/kobalt")
if(NOT EXISTS "${KOBALT_ROOT}/cmake/ReaderComponents.cmake")
  message(FATAL_ERROR "Run git submodule update --init external/kobalt first")
endif()
set(KOBALT_QT_MAJOR 6)
if(KOBALT_PROBE_HOST)
  find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
  add_executable(kobalt-probe main.cpp)
  target_compile_definitions(kobalt-probe PRIVATE KOBALT_PROBE_HOST)
else()
  include(../../mk/pocketbook-qt6.cmake)
  pocketbook_add_app(kobalt-probe SOURCES main.cpp)
endif()
include("${KOBALT_ROOT}/cmake/Strings.cmake")
include("${KOBALT_ROOT}/cmake/ReaderComponents.cmake")
kobalt_publication_library(kobalt-publication)
target_compile_options(kobalt-publication PRIVATE "-iquote${KOBALT_ROOT}/src")
target_include_directories(kobalt-probe PRIVATE "${KOBALT_ROOT}/src")
target_link_libraries(kobalt-probe PRIVATE kobalt-publication Qt6::Widgets Qt6::Gui Qt6::Core)
if(KOBALT_PROBE_HOST)
  enable_testing()
  execute_process(COMMAND "${Python3_EXECUTABLE}" -c
    "import runpy,sys; runpy.run_path(sys.argv[1])['book'](sys.argv[2], chapter_count=1, paragraphs=1)"
    "${KOBALT_ROOT}/tests/fixtures/reader_fixture.py" "${CMAKE_CURRENT_BINARY_DIR}/probe.epub"
    COMMAND_ERROR_IS_FATAL ANY)
  add_test(NAME kobalt-publication-widgets COMMAND kobalt-probe --self-test "${CMAKE_CURRENT_BINARY_DIR}/probe.epub")
  set_tests_properties(kobalt-publication-widgets PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
